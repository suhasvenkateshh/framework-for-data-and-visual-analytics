
import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt

from sklearn.preprocessing import MinMaxScaler, StandardScaler

data = {
    'Gender': ['Male', 'Female', 'Male', 'Male', 'Male'],
    'Married': ['Yes', 'No', 'Yes', np.nan, 'Yes'],
    'Dependents': ['0', '1', '3+', '2', '0'],
    'Education': ['Graduate', 'Not Graduate', 'Graduate',
                  'Graduate', 'Graduate'],
    'Self_Employed': ['No', 'Yes', 'No', 'No', 'No'],
    'ApplicantIncome': [5000, 3000, 4000, 6000, 5000],
    'CoapplicantIncome': [1500, 0, 2000, 1000, 1500],
    'LoanAmount': [150, 100, np.nan, 200, 150],
    'Loan_Amount_Term': [360, 360, 360, np.nan, 360],
    'Credit_History': [1, 1, np.nan, 1, 1],
    'Property_Area': ['Urban', 'Rural', 'Semiurban',
                      'Urban', 'Urban']
}

df = pd.DataFrame(data)

df.to_csv('/content/loan_dataset.csv', index=False)

print("Dataset created successfully!\n")
print(df)
print("\n" + "=" * 60)
print("INITIAL DATA OVERVIEW")
print("=" * 60)

df.info()

print("\nFirst 5 Rows:")
print(df.head())
print("\n" + "=" * 60)
print("MISSING VALUES BEFORE CLEANING")
print("=" * 60)

print(df.isnull().sum())

plt.figure(figsize=(10, 5))

sns.heatmap(
    df.isnull(),
    cbar=False,
    cmap="Blues"
)

plt.title("Missing Value Heatmap")
plt.xlabel("Columns")
plt.ylabel("Rows")
plt.show()

for col in ['Gender', 'Married', 'Dependents', 'Self_Employed']:
    df[col] = df[col].fillna(df[col].mode()[0])

df['LoanAmount'] = df['LoanAmount'].fillna(
    df['LoanAmount'].median()
)

df['Loan_Amount_Term'] = df['Loan_Amount_Term'].fillna(
    df['Loan_Amount_Term'].mode()[0]
)

df['Credit_History'] = df['Credit_History'].fillna(
    df['Credit_History'].mode()[0]
)
print("\n" + "=" * 60)
print("MISSING VALUES AFTER CLEANING")
print("=" * 60)

print(df.isnull().sum())

initial_rows = df.shape[0]

df.drop_duplicates(inplace=True)

removed_rows = initial_rows - df.shape[0]

print("\n" + "=" * 60)
print("DUPLICATE REMOVAL")
print("=" * 60)
print("Initial number of rows :", initial_rows)
print("Final number of rows   :", df.shape[0])
print("Duplicate rows removed :", removed_rows)

df['Dependents'] = df['Dependents'].replace('3+', 3)


df['Dependents'] = df['Dependents'].astype(int)

print("\n" + "=" * 60)
print("DATA TYPES AFTER CONVERSION")
print("=" * 60)
print(df.dtypes)

categorical_columns = [
    'Gender',
    'Married',
    'Education',
    'Self_Employed',
    'Property_Area'
]

for col in categorical_columns:
    df[col] = (
        df[col]
        .astype(str)
        .str.strip()
        .str.capitalize()
    )

print("\nCategorical values cleaned successfully.")

min_max_scaler = MinMaxScaler()
scale_columns = [
    'ApplicantIncome',
    'CoapplicantIncome',
    'LoanAmount'
]

df[scale_columns] = min_max_scaler.fit_transform(
    df[scale_columns]
)

print("\nMin-Max normalization completed successfully.")

standard_scaler = StandardScaler()
df[['Credit_History']] = standard_scaler.fit_transform(
    df[['Credit_History']]
)

print("\nNormalization and standardization completed successfully.")
print("\n" + "=" * 60)
print("FINAL CLEANED DATA")
print("=" * 60)
print(df)

print("\n" + "=" * 60)
print("FINAL DATA SUMMARY")
print("=" * 60)

df.info()
print("\n" + "=" * 60)
print("STATISTICAL SUMMARY")
print("=" * 60)

print(df.describe())
